```python
import asyncio, os
from claude_agent_sdk import ClaudeSDKClient, ClaudeAgentOptions
from telegram import Update
from telegram.ext import ApplicationBuilder, MessageHandler, ContextTypes, filters

TELEGRAM_TOKEN = os.environ["TELEGRAM_TOKEN"]
MON_USER_ID    = int(os.environ["MON_TELEGRAM_ID"])   # verrou : toi seule
REPO = "/Users/bettydeclety/Documents/TPDL/claude code dashboard"

async def on_message(update: Update, ctx: ContextTypes.DEFAULT_TYPE):
    if update.effective_user.id != MON_USER_ID:        # garde-fou
        return
    prompt = update.message.text
    await ctx.bot.send_chat_action(update.effective_chat.id, "typing")

    options = ClaudeAgentOptions(
        cwd=REPO,
        permission_mode="acceptEdits",                 # pas de blocage sur les edits
        allowed_tools=["Read", "Edit", "Write", "Bash", "Grep", "Glob"],
    )
    reply = []
    async with ClaudeSDKClient(options=options) as client:
        await client.query(prompt)
        async for msg in client.receive_response():
            # n'extraire que le texte final ; adapte selon le type de msg
            text = getattr(msg, "text", None) or str(msg)
            reply.append(text)

    out = "\n".join(reply)[-4000:]                      # limite Telegram = 4096 car.
    await update.message.reply_text(out or "(pas de réponse)")

app = ApplicationBuilder().token(TELEGRAM_TOKEN).build()
app.add_handler(MessageHandler(filters.TEXT & ~filters.COMMAND, on_message))
app.run_polling()
```

- **Verrouille l'accès à ton seul user ID** (le `if` ci-dessus) — sinon n'importe qui trouvant le bot pilote ton repo.

- La machine doit **rester allumée** ; pour que ça survive à un reboot, mets-le en `launchd` (Mac) ou sur un petit VPS.
